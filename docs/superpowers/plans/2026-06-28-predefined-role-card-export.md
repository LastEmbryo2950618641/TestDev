# 预定义角色卡导出与默认账号直连 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现按指定存档名导出刘悠与三胞胎四张预定义角色卡，并让默认账号确认时直接使用这些卡创建玩家卡和三张角色卡。

**Architecture:** 将导出逻辑拆成纯 Node 可测试转换模块、CLI/浏览器桥接工具、运行时预定义卡接入三层。纯转换层不依赖 DOM、dzmm 或 sql.js，负责从角色状态构建卡、成人化、性经验归零、渲染 JSON/JS；读取 slot 的动作放在工具入口中，在浏览器上下文读取 `control-rpg-sqlite-${slot}`，Node 侧接收 raw 并写文件。

**Tech Stack:** Node.js CommonJS、浏览器 `window.GameModules.sqliteSave` / `dzmm.kv` / `localStorage`、现有 `publish/predefined-role-cards.js`、现有无框架 Node 测试风格（`assert` + `vm`）。

## Global Constraints

- 不修改、不读取 `publish/config/default-existing-profile.*` 作为资料源。
- 工具以指定存档名/slot 名为主入口，不能要求用户先手动取得角色状态 JSON。
- 生成四张卡：刘悠为玩家卡，刘思瑶、刘思琪、刘思怡为角色卡。
- 三胞胎统一成人化为 18 岁。
- 性经验次数全部归 0。
- 除年龄成人化与性经验次数归 0 外，尽量保持当前存档角色状态中的信息。
- 默认账号确认时直接使用预定义卡创建 `player-self` 与三张关系角色卡，不再请求 AI 生成这些卡。
- 不创建 git commit。
- 不清理或回滚当前工作区中无关未提交改动。

```js
#!/usr/bin/env node
const fs = require('fs');
const path = require('path');

const TARGET_PEOPLE = [
  { name: '刘悠', slug: 'liu-you', isPlayer: true },
  { name: '刘思瑶', slug: 'liu-siyao', isPlayer: false },
  { name: '刘思琪', slug: 'liu-siqi', isPlayer: false },
  { name: '刘思怡', slug: 'liu-siyi', isPlayer: false },
];

function clone(value) {
  return value === undefined ? undefined : JSON.parse(JSON.stringify(value));
}

function targetByName(name = '') {
  return TARGET_PEOPLE.find((item) => item.name === name) || null;
}

function replaceAgeText(text) {
  return typeof text === 'string' ? text.replace(/15\s*岁/gu, '18岁') : text;
}

function deepMap(value, mapper) {
  const mapped = mapper(value);
  if (mapped !== value) return mapped;
  if (Array.isArray(value)) return value.map((item) => deepMap(item, mapper));
  if (value && typeof value === 'object') {
    return Object.fromEntries(Object.entries(value).map(([key, item]) => [key, deepMap(item, (inner) => {
      if ((key === 'sexualExperienceCount' || key === '性经验次数') && typeof inner === 'number') return 0;
      return mapper(inner);
    })]));
  }
  return value;
}

function normalizeAdultTriplet(card) {
  if (!['刘思瑶', '刘思琪', '刘思怡'].includes(card.name)) return card;
  const next = deepMap(card, replaceAgeText);
  next.age = 18;
  next.role = replaceAgeText(next.role || '18岁三胞胎妹妹');
  next.detail = replaceAgeText(next.detail || '');
  next.values = next.values || {};
  next.values.age = 18;
  return next;
}

function zeroSexualExperienceCounts(value) {
  const next = deepMap(value, (item) => item);
  function walk(obj) {
    if (!obj || typeof obj !== 'object') return;
    Object.keys(obj).forEach((key) => {
      if ((key === 'sexualExperienceCount' || key === '性经验次数') && typeof obj[key] === 'number') obj[key] = 0;
      else walk(obj[key]);
    });
  }
  walk(next);
  if (next.values?.intimacy && typeof next.values.intimacy === 'object') next.values.intimacy.sexualExperienceCount = 0;
  if (next.intimacy && typeof next.intimacy === 'object') next.intimacy.sexualExperienceCount = 0;
  return next;
}

function extractCharacterStates(input) {
  if (Array.isArray(input)) return input;
  if (!input || typeof input !== 'object') return [];
  if (Array.isArray(input.characterStates)) return input.characterStates;
  if (input.characterStates && typeof input.characterStates === 'object') return Object.values(input.characterStates);
  if (input.fallbackState?.characterStates && typeof input.fallbackState.characterStates === 'object') return Object.values(input.fallbackState.characterStates);
  if (input.main?.rpgStates && typeof input.main.rpgStates === 'object') return Object.values(input.main.rpgStates);
  return [];
}

function metricsToInitialMetrics(metrics = {}, fallback = {}) {
  if (fallback.initialMetrics?.emotions || fallback.initialMetrics?.playerFeelings) return clone(fallback.initialMetrics);
  const entries = (obj) => Object.entries(obj || {}).map(([key, value]) => ({ key, value: Number(value) || 0 }));
  return { emotions: entries(metrics.emotions), playerFeelings: entries(metrics.playerFeelings) };
}

function buildCardFromState(state = {}) {
  const profile = clone(state.profile || state) || {};
  const target = targetByName(profile.name || state.name);
  if (!target) throw new Error(`非目标角色：${profile.name || state.name || 'unknown'}`);
  const values = clone(state.values || profile.values || {});
  const card = {
    ...profile,
    id: target.isPlayer ? 'player-self' : (profile.id || state.id || target.slug),
    name: target.name,
    age: profile.age ?? values.age ?? '',
    work: profile.work || state.worldTag || values.world_tag || '2026 现代都市现实世界',
    values,
    items: Array.isArray(profile.items) && profile.items.length ? clone(profile.items) : clone(values.items || []),
    wearing: Array.isArray(profile.wearing) && profile.wearing.length ? clone(profile.wearing) : clone(values.wearing || []),
    factions: Array.isArray(profile.factions) && profile.factions.length ? clone(profile.factions) : clone(values.factions || []),
    force_positions: Array.isArray(profile.force_positions) && profile.force_positions.length ? clone(profile.force_positions) : clone(values.force_positions || []),
    bodyStatus: clone(profile.bodyStatus || values.bodyStatus || {}),
    intimacy: clone(profile.intimacy || values.intimacy || {}),
    initialMetrics: metricsToInitialMetrics(state.metrics || {}, profile),
    worldAttributes: clone(profile.worldAttributes || { worldTag: state.worldTag || profile.work || '2026 现代都市现实世界', source: 'predefined-export', fields: [] }),
    roleCardFieldReasons: clone(profile.roleCardFieldReasons || {}),
    rpgFieldReasons: clone(profile.rpgFieldReasons || {}),
    roleCard: true,
    roleCardSource: 'predefined',
    roleCardUpdatedAt: new Date().toISOString(),
  };
  if (target.isPlayer) card.isPlayer = true;
  return zeroSexualExperienceCounts(normalizeAdultTriplet(card));
}

function buildExportBundle(states = []) {
  const cardsByName = new Map(extractCharacterStates(states).map((state) => [state.profile?.name || state.name, state]));
  const missing = TARGET_PEOPLE.filter((target) => !cardsByName.has(target.name));
  if (missing.length) throw new Error(`存档缺少目标角色：${missing.map((item) => item.name).join('、')}`);
  return TARGET_PEOPLE.map((target) => {
    const card = buildCardFromState(cardsByName.get(target.name));
    const json = `${JSON.stringify(card, null, 2)}\n`;
    const js = renderCardJs(target.slug, card);
    return { slug: target.slug, name: target.name, card, json, js };
  });
}

function renderCardJs(slug, card) {
  return `window.GameModules = window.GameModules || {};\nwindow.GameModules.predefinedRoleCardData = window.GameModules.predefinedRoleCardData || {};\nwindow.GameModules.predefinedRoleCardData['${slug}'] = ${JSON.stringify(card, null, 2)};\n`;
}

function writeExportBundle(bundle, outDir) {
  fs.mkdirSync(outDir, { recursive: true });
  const written = [];
  bundle.forEach((item) => {
    const jsonPath = path.join(outDir, `${item.slug}.json`);
    const jsPath = path.join(outDir, `${item.slug}.js`);
    fs.writeFileSync(jsonPath, item.json, 'utf8');
    fs.writeFileSync(jsPath, item.js, 'utf8');
    written.push(jsonPath, jsPath);
  });
  return written;
}

module.exports = {
  TARGET_PEOPLE,
  extractCharacterStates,
  buildCardFromState,
  buildExportBundle,
  renderCardJs,
  writeExportBundle,
  zeroSexualExperienceCounts,
  normalizeAdultTriplet,
};
```

```js
#!/usr/bin/env node
const fs = require('fs');
const path = require('path');
const exporter = require('./predefined-role-card-exporter');

function parseArgs(argv = process.argv.slice(2)) {
  const args = { slot: 'slot-1', out: path.join('publish', 'predefined-role-cards'), publish: 'publish', dryRun: false, rawFile: '', printBrowserSnippet: false };
  for (let i = 0; i < argv.length; i += 1) {
    const item = argv[i];
    if (item === '--slot') args.slot = argv[++i] || args.slot;
    else if (item === '--out') args.out = argv[++i] || args.out;
    else if (item === '--publish') args.publish = argv[++i] || args.publish;
    else if (item === '--raw-file') args.rawFile = argv[++i] || '';
    else if (item === '--dry-run') args.dryRun = true;
    else if (item === '--print-browser-snippet') args.printBrowserSnippet = true;
    else if (item === '-h' || item === '--help') args.help = true;
    else throw new Error(`未知参数：${item}`);
  }
  if (!args.out) args.out = path.join(args.publish, 'predefined-role-cards');
  return args;
}

function usage() {
  return [
    '用法:',
    '  node "tools/export-predefined-role-cards.js" --slot "slot-1"',
    '  node "tools/export-predefined-role-cards.js" --print-browser-snippet --slot "slot-1"',
    '  node "tools/export-predefined-role-cards.js" --raw-file "userInformation/save/control-rpg-sqlite-slot-1.raw"',
    '',
    '说明:',
    '  Node 无法直接访问浏览器 dzmm.kv 时，先运行 --print-browser-snippet，把输出粘贴到游戏页面控制台。',
    '  控制台脚本会按 slot 读取 control-rpg-sqlite-${slot} 并下载 raw 文件。',
    '  再用 --raw-file 指向下载的 raw 文件生成 publish/predefined-role-cards/*.json/*.js。',
  ].join('\n');
}

function browserSnippet(slot = 'slot-1') {
  return `(async()=>{const slot=${JSON.stringify(slot)};const key=\`control-rpg-sqlite-\${slot}\`;let raw=null;if(window.GameModules?.sqliteSave?.readRaw)raw=await window.GameModules.sqliteSave.readRaw(slot);if(!raw&&window.dzmm?.kv)raw=(await window.dzmm.kv.get(key))?.value||null;if(!raw)try{raw=localStorage.getItem(key)}catch(_){}if(!raw)throw new Error('未找到存档：'+key);const blob=new Blob([raw],{type:'text/plain;charset=utf-8'});const a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download=key+'.raw';a.click();setTimeout(()=>URL.revokeObjectURL(a.href),1000);console.log('已导出',key,'长度',raw.length);})();`;
}

async function statesFromSqliteBase64(raw, sqlJsFactory) {
  const initSqlJs = sqlJsFactory || global.initSqlJs;
  if (!initSqlJs) throw new Error('raw 看起来不是 JSON；如它是 SQLite base64，请提供 sql.js 运行环境或先用浏览器 raw 导出后安装 sql.js。');
  const SQL = await initSqlJs();
  const bytes = Uint8Array.from(Buffer.from(String(raw), 'base64'));
  const db = new SQL.Database(bytes);
  const rows = [];
  const stmt = db.prepare('SELECT state_json FROM character_state ORDER BY created_at');
  while (stmt.step()) rows.push(JSON.parse(stmt.getAsObject().state_json));
  stmt.free();
  db.close();
  return rows;
}

async function statesFromRaw(raw, sqlJsFactory = null) {
  const text = String(raw || '').trim();
  if (!text) throw new Error('raw 存档为空');
  if (text.startsWith('{') || text.startsWith('[')) return exporter.extractCharacterStates(JSON.parse(text));
  return statesFromSqliteBase64(text, sqlJsFactory);
}

async function readRawBySlot(_slot) {
  throw new Error('当前 Node 环境不能直接读取浏览器 dzmm.kv/localStorage。请先运行：node "tools/export-predefined-role-cards.js" --print-browser-snippet --slot "slot-1"，在游戏页面控制台粘贴输出脚本下载 raw 文件，再用 --raw-file 导出。');
}

async function main(argv = process.argv.slice(2)) {
  const args = parseArgs(argv);
  if (args.help) { console.log(usage()); return; }
  if (args.printBrowserSnippet) { console.log(browserSnippet(args.slot)); return; }
  const raw = args.rawFile ? fs.readFileSync(args.rawFile, 'utf8') : await readRawBySlot(args.slot);
  const states = await statesFromRaw(raw);
  const bundle = exporter.buildExportBundle(states);
  if (args.dryRun) {
    console.log(JSON.stringify(bundle.map((item) => ({ slug: item.slug, name: item.name, bytes: item.json.length })), null, 2));
    return;
  }
  const written = exporter.writeExportBundle(bundle, args.out);
  written.forEach((file) => console.log(`写入 ${file}`));
}

if (require.main === module) {
  main().catch((err) => {
    console.error(err.message || err);
    process.exit(1);
  });
}

module.exports = { parseArgs, usage, browserSnippet, statesFromRaw, statesFromSqliteBase64, main };
```
